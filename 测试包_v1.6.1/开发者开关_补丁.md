## 2. 补丁正文（整段复制）

```js
  /* ============================================================
     DEV 开发者开关（?dev=1）· 默认关闭 · 只认 URL 参数
     ------------------------------------------------------------
     用途：复现问题时快速跳层 / 回满血 / 加结晶 / 切换永久天赋。
     入口：浏览器地址栏在 .html 后面加 ?dev=1，例如
           file:///D:/其他/TRAE宝/奥术工程师/dist/emberfall_v1.6.1/烬_v1.6.1.html?dev=1
     约定：不带 ?dev=1 时整段直接 return，正式玩家零影响。
     回滚：删除本段（从本注释块到下面 `})();` 为止）即可。
     ============================================================ */
  (function installDevSwitch() {
    /** 是否启用：只认 URL 参数，不读存档、不写存档开关 */
    let devOn = false;
    try {
      devOn = new URLSearchParams(window.location.search).get('dev') === '1';
    } catch (err) {
      devOn = false;
    }
    if (!devOn) return;

    /** 天赋预设档：0 清空 / 1 生存 / 2 输出 / 3 经济 / 4 全满 */
    const DEV_PRESETS = ['清空', '生存满', '输出满', '经济满', '全部满'];
    let devPreset = 0;

    /* 右下角只读面板：pointer-events:none，不挡鼠标、不影响玩法 */
    const devPanel = document.createElement('div');
    devPanel.id = 'devPanel';
    devPanel.style.cssText = [
      'position:fixed', 'right:10px', 'bottom:10px', 'z-index:9991',
      'pointer-events:none', 'background:rgba(6,12,20,.84)',
      'border:1px solid #2f6f88', 'border-radius:8px', 'padding:8px 10px',
      'font:12px/1.6 Consolas,monospace', 'color:#bfe9ff',
      'white-space:pre', 'letter-spacing:.02em'
    ].join(';');
    document.body.appendChild(devPanel);

    /**
     * 刷新右下角面板上的实时状态（每 600ms 一次，开销可忽略）
     * @returns {void}
     */
    function devRefresh() {
      const live = state && state.mode === 'playing' && state.player;
      const head = 'DEV · ?dev=1\n'
        + '层： [  -1    ]  +1\n'
        + '血： -  回满\n'
        + '钱： =  +1000 结晶\n'
        + '天赋：0  预设『' + DEV_PRESETS[devPreset] + '』\n';
      devPanel.textContent = live
        ? head + '现在：第 ' + state.floor + ' 层 | HP '
          + Math.round(state.player.hp) + '/' + Math.round(state.player.hpMax)
          + ' | 结晶 ' + (Number(state.meta.crystals) || 0)
        : head + '现在：未开始（先点开始探索）';
    }

    /**
     * DEV 跳层：完全复用正式 generateFloor，不额外造地图、不改随机流规则
     * @param {number} targetFloor 目标层（1–20，超界自动夹取）
     * @returns {boolean} 是否真的跳了
     */
    function devJumpTo(targetFloor) {
      if (!state || state.mode !== 'playing') {
        showToast('DEV：先开始一局再跳层', 'warn');
        return false;
      }
      const target = Math.max(1, Math.min(20, Math.floor(Number(targetFloor) || 1)));
      if (target === state.floor) {
        showToast('DEV：已经在这一层了', 'warn');
        return false;
      }
      state.floor = target;
      generateFloor(target);          // 正式楼层生成：房间 / 敌人 / 首领 / 场景烘焙全走它
      placePlayerAtSpawn();
      state.player.hp = state.player.hpMax;
      state.player.aether = state.player.aetherMax;
      state.floorNoHit = true;        // 跳层不记"无伤层"惩罚
      resetDeepHealBossBucket();
      state.meta.bestFloor = Math.max(Number(state.meta.bestFloor) || 1, target);
      saveMeta();
      updateHUD();
      renderDeck();
      drawMinimap();
      updateTitleStats();
      showToast('DEV：已跳到第 ' + target + ' 层'
        + (isRegionBossFloor(target) ? '（首领层）' : ''), 'success');
      devRefresh();
      return true;
    }

    /**
     * DEV 回满生命与奥能（不改变上限）
     * @returns {boolean} 是否生效
     */
    function devHeal() {
      if (!state || state.mode !== 'playing' || !state.player) {
        showToast('DEV：先开始一局', 'warn');
        return false;
      }
      state.player.hp = state.player.hpMax;
      state.player.aether = state.player.aetherMax;
      updateHUD();
      showToast('DEV：生命 / 奥能已回满', 'success');
      devRefresh();
      return true;
    }

    /**
     * DEV 加奥术结晶（只写 meta，用于测试天赋升级）
     * @param {number} amount 增量，默认 1000
     * @returns {number} 加完后的余额
     */
    function devAddCrystals(amount) {
      if (!state || !state.meta) return 0;
      const add = Math.floor(Number(amount) || 1000);
      const cur = Math.max(0, Number(state.meta.crystals) || 0);
      if (window.__talentDebug && typeof window.__talentDebug.setCrystals === 'function') {
        window.__talentDebug.setCrystals(cur + add);   // 走游戏自己的写档 + 刷新
      } else {
        state.meta.crystals = cur + add;
        saveMeta();
        renderTalents();
      }
      updateTitleStats();
      showToast('DEV：结晶 +' + add + '（现有 ' + state.meta.crystals + '）', 'success');
      devRefresh();
      return state.meta.crystals;
    }

    /**
     * DEV 循环切换永久天赋预设（0 清空 → 1 生存 → 2 输出 → 3 经济 → 4 全满）
     * @returns {string} 切换后的预设名
     */
    function devCycleTalentPreset() {
      devPreset = (devPreset + 1) % DEV_PRESETS.length;
      const dbg = window.__talentDebug;
      if (!dbg || typeof dbg.resetTalents !== 'function') {
        showToast('DEV：天赋接口未就绪', 'warn');
        return DEV_PRESETS[devPreset];
      }
      dbg.resetTalents();
      if (devPreset > 0) {
        const cats = devPreset === 1 ? ['survival']
          : devPreset === 2 ? ['output']
            : devPreset === 3 ? ['economy']
              : ['survival', 'output', 'economy', 'unlock'];
        TALENT_CONFIG.forEach(function (t) {
          if (cats.indexOf(t.cat) >= 0) dbg.setLevel(t.id, t.maxLevel);
        });
      }
      /* 天赋换了，上限也变了：顺手回满，方便直接看效果 */
      if (state && state.player) {
        state.player.hp = state.player.hpMax;
        state.player.aether = state.player.aetherMax;
      }
      updateHUD();
      updateTitleStats();
      showToast('DEV：天赋预设 → ' + DEV_PRESETS[devPreset], 'success');
      devRefresh();
      return DEV_PRESETS[devPreset];
    }

    /**
     * DEV 键盘入口（捕获阶段，选的都是玩法未占用的键）
     * @param {KeyboardEvent} event 键盘事件
     * @returns {void}
     */
    function devKeyDown(event) {
      const target = event.target;
      if (target && /^(INPUT|TEXTAREA|SELECT)$/.test(target.tagName || '')) return;
      const key = String(event.key || '');
      const keyLower = key.toLowerCase();
      if (key === '[') { event.preventDefault(); event.stopPropagation(); devJumpTo((state ? state.floor : 1) - 1); }
      else if (key === ']' || key === '}') { event.preventDefault(); event.stopPropagation(); devJumpTo((state ? state.floor : 1) + 1); }
      else if (key === '-' || key === '_') { event.preventDefault(); event.stopPropagation(); devHeal(); }
      else if (key === '=' || key === '+') { event.preventDefault(); event.stopPropagation(); devAddCrystals(1000); }
      else if (key === '0') { event.preventDefault(); event.stopPropagation(); devCycleTalentPreset(); }
      else if (keyLower === 'p') { /* 保留：未来可扩展，暂不占用 */ }
    }

    window.addEventListener('keydown', devKeyDown, true);
    setInterval(devRefresh, 600);
    devRefresh();

    /* 对外只读接口：自动化测试用；正式玩家不调用也不影响 */
    window.__devMode = {
      jumpTo: devJumpTo,
      heal: devHeal,
      addCrystals: devAddCrystals,
      cycleTalentPreset: devCycleTalentPreset,
      presetName: function () { return DEV_PRESETS[devPreset]; },
      info: function () {
        return {
          enabled: true,
          floor: state ? state.floor : 0,
          mode: state ? state.mode : 'none',
          preset: DEV_PRESETS[devPreset]
        };
      }
    };
  })();
```

## 3. 用法（给使用者）

1. 打开浏览器（Edge），地址栏整段粘贴（路径按你机器实际位置替换）：
   `file:///D:/其他/TRAE宝/奥术工程师/dist/emberfall_v1.6.1/烬_v1.6.1.html?dev=1`
2. 点「开始探索」进入局内后，右下角出现小面板 `DEV · ?dev=1`。
3. 按键：

| 按键 | 作用 |
| --- | --- |
| `[` | 上一层 |
| `]` | 下一层（连按可快速到 6 / 12 / 18 层验首领） |
| `-` | 生命 + 奥能回满 |
| `=` | 结晶 +1000 |
| `0` | 循环永久天赋预设：清空 → 生存满 → 输出满 → 经济满 → 全部满 |

4. 去掉地址栏的 `?dev=1` 再打开，一切恢复正式行为（面板、快捷键、`__devMode` 都不存在）。
