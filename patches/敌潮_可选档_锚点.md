```js
/* ===== 可选敌潮（玩家开关，默认 off）· 三档 ==========================
   · 只加"每房敌人数量"：本段永远不许出现 damageMul / hpMul 这类伤害或血量乘区，
     以后谁想把它接到伤害上，先回去看批次叠加矩阵 §4 的验收口径
   · 与 DENSITY_CONFIG（常驻密度）是两层，互不替代，可叠加
   · 文案口径："更多敌人、更多掉落"；任何界面文案禁止出现"困难模式"
   · 默认值：TIDE_SETTINGS.tier = 'off'；设置页只改这个字段（setTideTier 负责持久化）
   ==================================================================== */
const TIDE_TIER_CONFIG = {
  tiers: [
    { id: 't1', label: '更多敌人',        addEarly: 0, addLate: 1 },
    { id: 't2', label: '更多敌人 · 涌动', addEarly: 1, addLate: 2 },
    { id: 't3', label: '更多敌人 · 压境', addEarly: 2, addLate: 2 }
  ],
  noAddUntil: 2,     // F1–2 不加（与常驻密度同闸门，保护新手关）
  earlyUntil: 5,     // F3–5 用 addEarly
  eliteAdd: 1,       // 精英房当前 ≤eliteMaxBase 只时 +1
  eliteMaxBase: 5,
  roomCap: 12
};

/* 默认 off。设置页的开关通过 setTideTier() 写这里；持久化键固定 'emberfall.tideTier' */
const TIDE_SETTINGS = {
  tier: (function () {
    try { return localStorage.getItem('emberfall.tideTier') || 'off'; } catch (e) { return 'off'; }
  })()
};

/**
 * 设置敌潮档位并持久化（设置页唯一写入口）。
 * @param {string} id 'off' | 't1' | 't2' | 't3'
 * @returns {string} 实际生效的档位
 */
function setTideTier(id) {
  const ok = id === 'off' || TIDE_TIER_CONFIG.tiers.some((t) => t.id === id);
  TIDE_SETTINGS.tier = ok ? id : 'off';
  try { localStorage.setItem('emberfall.tideTier', TIDE_SETTINGS.tier); } catch (e) { /* 隐私模式忽略 */ }
  return TIDE_SETTINGS.tier;
}

/**
 * 可选敌潮：在常驻密度之后再加一层"数量"，不碰伤害与血量。
 * @param {Object} room 房间对象
 * @param {number} count 当前敌人数（已含 DENSITY_CONFIG 与 P2 进阶）
 * @param {number} floorNumber 层数
 * @returns {number} 加完后的敌人数
 */
function tideTierCount(room, count, floorNumber) {
  const id = TIDE_SETTINGS.tier;
  if (id === 'off') return count;
  if (!room || (room.type !== 'battle' && room.type !== 'elite')) return count;
  if (floorNumber <= TIDE_TIER_CONFIG.noAddUntil) return count;
  const tier = TIDE_TIER_CONFIG.tiers.find((t) => t.id === id);
  if (!tier) return count;
  if (room.type === 'elite') {
    if (count <= TIDE_TIER_CONFIG.eliteMaxBase) count += TIDE_TIER_CONFIG.eliteAdd;
  } else {
    count += floorNumber <= TIDE_TIER_CONFIG.earlyUntil ? tier.addEarly : tier.addLate;
  }
  return Math.max(1, Math.min(TIDE_TIER_CONFIG.roomCap, count));
}
```

## 4. 默认值与设置页入口（整合线接）

- **默认值位置**：`TIDE_SETTINGS.tier`，初始 `'off'`（上块里已写）。
- **设置页唯一写入口**：`setTideTier('off' | 't1' | 't2' | 't3')`，内部负责 `localStorage`。
- **建议 UI 形态**：设置页「更多敌人」下拉/三按钮，**默认不选（off）**；开启后默认停 `t1`。
- **不要**把 `TIDE_SETTINGS.tier` 写进难度配置，也不要和 `DIFFICULTY_CONFIG` 混在一起。

## 5. 验收断言（贴完先跑这三条）

1. **数量断言**：同一种子、第 6 层普通战斗房，`t1` 的 `room.enemyCount` 应比 `off` **+1**；
   `t2`/`t3` 为 **+2**。F1–2 两者必须**完全相等**（`noAddUntil`）。
2. **解耦断言**：同一房间下，`difficultyCfg().damageMul` 与 `difficultyCfg().hpMul` 在
   `off` / `t1` / `t2` / `t3` 四种状态下**逐位相同**（`console.log` 对比即可）。
3. **边界断言**：在 `t3` + F10 精英房下，`room.enemyCount` 不得超过 `TIDE_TIER_CONFIG.roomCap`（12）。
